// app/api/images/[month]/[id]/[type]/route.ts

export async function GET(
    request: NextRequest,
    { params }: { params: Promise<{ month: string; id: string; type: string }> }
) {
    const { month, id, type } = await params;

    // 1. 路径解析（支持三种内容类型）
    let baseDir = '';
    const subjectMatch = month.match(/^(\d{4})-subject-(.+)$/);
    if (subjectMatch) {
        const [_, year, name] = subjectMatch;
        baseDir = path.join(process.cwd(), 'public', 'content', year, 'subject', decodeURIComponent(name), id);
    }
    // ... sleepingMatch, monthMatch 处理逻辑

    // 2. 文件类型映射
    let filePath = '';
    if (type === 'card' || type === 'thumbnail') {
        filePath = path.join(baseDir, `${id}.png`);
    } else if (type === 'cover') {
        filePath = path.join(baseDir, 'pic', 'cover.jpg');
    }
    // ... cover-thumbnail 处理

    // 3. 响应与缓存
    try {
        const imageBuffer = await fs.promises.readFile(filePath);
        return new NextResponse(imageBuffer, {
            headers: {
                'Content-Type': contentType,
                'Cache-Control': 'public, max-age=31536000, immutable',  // 1年缓存
            },
        });
    } catch (error) {
        return new NextResponse('Image not found', { status: 404 });
    }
}

### 3.2 Framer Motion 与物理引擎的使用

#### 用户体验痛点分析

```markdown
传统网页浏览痛点:
1. 静态展示 - 书籍卡片固定不动，缺乏生命力
2. 线性导航 - 只能点击下一页，探索效率低
3. 缺乏反馈 - 操作后无视觉反馈，交互枯燥
4. 信息过载 - 大量书籍同时展示，难以聚焦
```

#### 解决方案：物理引擎动画

## 四、总结

### 技术决策合理性

| 决策 | 选型 | 理由 |
|-----|------|-----|
| 状态管理 | Zustand | 轻量、TypeScript 友好、持久化支持 |
| 渲染策略 | SSG + ISR | 兼顾性能与数据更新 |
| 图片服务 | Cloudflare R2 + CDN | 自动优化、全球分发 |
| 动画引擎 | Framer Motion | 声明式、物理引擎、自然交互 |
| 数据存储 | 文件系统 (JSON) | 简化部署、Git 版本控制 |

### 架构优势

1. **高内聚低耦合**：各层职责清晰，通过接口通信
2. **可测试性**：Zustand store 可单独测试
3. **可扩展性**：新增内容类型只需添加目录
4. **高性能**：静态生成 + CDN 缓存
5. **好体验**：物理引擎提供自然交互
